\node[process] (filter) at (0,8.2) {Apply RSSI filtering\\(moving average)};

\node[io] (output) at (0,-6.6) {Output: Final position\\(X, Y) with timestamp};

\node[startstop] (end) at (0,-9.6) {End};

\draw[arrow] (start) -- (input);

\draw[arrow] (valid) -- node[above, font=\tiny] {No} (discard);

\draw[arrow] (valid) -- node[right, font=\tiny] {Yes} (method);

\draw[arrow] (ml_output) -- ++(-2,0) |- (validate);

\draw[arrow] (tri_convert) -- (tri_check);
\draw[arrow] (tri_check) -- node[right, font=\tiny] {Yes} (tri_calc);
\draw[arrow] (tri_calc) -- (tri_output);
\draw[arrow] (tri_output) -- ++(2,0) |- (validate);
\draw[arrow] (tri_check) -- node[above, font=\tiny] {No} (tri_error);

\draw[arrow] (output) -- (store);
\draw[arrow] (store) -- (end);
